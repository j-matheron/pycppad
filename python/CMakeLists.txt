# --- LIBRARY --- #

FIND_PACKAGE(Boost REQUIRED)

# Name of the Python library
SET(PYTHON_LIB_NAME ${PROJECT_NAME}_pywrap)
message("give the lib name " ${PYTHON_LIB_NAME})
# necessary ?
SET(PYTHON_LIB_NAME ${PYTHON_LIB_NAME} PARENT_SCOPE)

MAKE_DIRECTORY("${${PROJECT_NAME}_BINARY_DIR}/python/${PROJECT_NAME}")

ADD_CUSTOM_TARGET(${PROJECT_NAME}_python)
SET_TARGET_PROPERTIES(${PROJECT_NAME}_python PROPERTIES EXCLUDE_FROM_DEFAULT_BUILD True)


SET(${PYTHON_LIB_NAME}_HEADERS
    include/${PROJECT_NAME}/fwd.hpp
    include/${PROJECT_NAME}/ad.hpp
    include/${PROJECT_NAME}/independent.hpp
    include/${PROJECT_NAME}/ad_fun.hpp
    include/${PROJECT_NAME}/cast.hpp
    include/${PROJECT_NAME}/cppad.hpp
    include/${PROJECT_NAME}/cppad-scalar.hpp
    include/${PROJECT_NAME}/utils/scope.hpp
)

SET(${PYTHON_LIB_NAME}_SOURCES
   ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp
   ${CMAKE_SOURCE_DIR}/src/cppad.cpp
   ${CMAKE_SOURCE_DIR}/src/expose-type-double.cpp
   ${CMAKE_SOURCE_DIR}/src/expose-type-double-row-major.cpp
)

ADD_LIBRARY(${PYTHON_LIB_NAME} MODULE ${${PYTHON_LIB_NAME}_HEADERS} ${${PYTHON_LIB_NAME}_SOURCES})

IF(SUFFIX_SO_VERSION)
  SET_TARGET_PROPERTIES(${PYTHON_LIB_NAME} PROPERTIES SOVERSION ${PROJECT_VERSION})
ENDIF(SUFFIX_SO_VERSION)

# target_include_directories
TARGET_INCLUDE_DIRECTORIES(${PYTHON_LIB_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

TARGET_INCLUDE_DIRECTORIES(
  ${PYTHON_LIB_NAME}
  SYSTEM
  PRIVATE ${PYTHON_INCLUDE_DIRS})

TARGET_LINK_LIBRARIES(
 ${PYTHON_LIB_NAME} PUBLIC ${cppad_LIBRARY} eigenpy::eigenpy)

TARGET_LINK_BOOST_PYTHON(${PYTHON_LIB_NAME} PUBLIC)

# BOOST_PYTHON_MEMBER_FUNCTION_OVERLOADS spews conversion warnings from int to long unsigned int.
# Unfortunately, using literals does not work in a macro. As such, this turns them off for the entire wrapper:
IF(NOT WIN32)
  TARGET_COMPILE_OPTIONS(${PYTHON_LIB_NAME} PRIVATE "-Wno-conversion")
ENDIF()
IF(IS_ABSOLUTE ${PYTHON_SITELIB})
  SET(${PYTHON_LIB_NAME}_INSTALL_DIR ${PYTHON_SITELIB}/${PROJECT_NAME})
ELSE()
  SET(${PYTHON_LIB_NAME}_INSTALL_DIR ${CMAKE_INSTALL_PREFIX}/${PYTHON_SITELIB}/${PROJECT_NAME})
ENDIF()
IF(WIN32)
  TARGET_COMPILE_DEFINITIONS(${PYTHON_LIB_NAME} PRIVATE -DNOMINMAX)
  TARGET_LINK_LIBRARIES(${PYTHON_LIB_NAME} PUBLIC ${PYTHON_LIBRARY})
ENDIF(WIN32)

ADD_DEPENDENCIES(${PROJECT_NAME}_python ${PYTHON_LIB_NAME})

ADD_HEADER_GROUP(${PYTHON_LIB_NAME}_HEADERS)
ADD_SOURCE_GROUP(${PYTHON_LIB_NAME}_SOURCES)

SET_TARGET_PROPERTIES(
    ${PYTHON_LIB_NAME}
    PROPERTIES
        PREFIX ""
        SUFFIX "${PYTHON_EXT_SUFFIX}"
        OUTPUT_NAME "${PYTHON_LIB_NAME}"
        LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/python/${PROJECT_NAME}"
        # On Windows, shared library are treated as binary
        RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/python/${PROJECT_NAME}"
)

IF(UNIX)
  GET_RELATIVE_RPATH(
    ${${PYTHON_LIB_NAME}_INSTALL_DIR}
    ${PYTHON_LIB_NAME}_INSTALL_RPATH
    )
    set_target_properties(
    ${PYTHON_LIB_NAME}
    PROPERTIES INSTALL_RPATH "${${PYTHON_LIB_NAME}_INSTALL_RPATH}"
    )
ENDIF()

INSTALL(
    TARGETS ${PYTHON_LIB_NAME}
    EXPORT ${TARGETS_EXPORT_NAME}
    DESTINATION ${${PYTHON_LIB_NAME}_INSTALL_DIR}
)

# --- INSTALL SCRIPTS
SET(PYTHON_FILES
  __init__.py
  windows_dll_manager.py
  )

FOREACH(python ${PYTHON_FILES})
  PYTHON_BUILD(${PROJECT_NAME} ${python})
  INSTALL(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/pycppad/${python}"
    DESTINATION ${${PYTHON_LIB_NAME}_INSTALL_DIR})
ENDFOREACH(python)
